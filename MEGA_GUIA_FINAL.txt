def processar_mensagem(texto_entrada, numero_usuario=None):
    """
    Processa mensagem e retorna resposta
    """
    
    texto_entrada = texto_entrada.lower().strip()
    
    # Saudações
    if any(word in texto_entrada for word in ['oi', 'olá', 'opa', 'hey', 'bom dia']):
        return "Olá! 👋 Bem-vindo! Como posso ajudar?"
    
    # Informações
    elif 'horário' in texto_entrada or 'funciona' in texto_entrada:
        return "📅 Funcionamos:\n• Seg-Sex: 09:00-18:00\n• Sábado: 10:00-14:00"
    
    elif 'preço' in texto_entrada or 'custa' in texto_entrada or 'valor' in texto_entrada:
        return "💰 Preços:\n• Plano A: R$ 50\n• Plano B: R$ 100\n• Plano C: R$ 150"
    
    elif 'endereço' in texto_entrada or 'localização' in texto_entrada:
        return "📍 Endereço:\nRua Principal, 123\nCidade, Estado\n☎️ (85) 3456-7890"
    
    # Menu
    elif 'menu' in texto_entrada or 'ajuda' in texto_entrada:
        return """📋 Menu Principal:
1️⃣ Horário de funcionamento
2️⃣ Preços
3️⃣ Endereço
4️⃣ Falar com atendente
5️⃣ Consultoria

Digite: 'horário', 'preço', 'endereço', 'atendente' ou 'consultoria'"""
    
    elif 'atendente' in texto_entrada or 'falar' in texto_entrada:
        return "👤 Conectando com atendente...\nTempo: 2 minutos"
    
    elif 'consultoria' in texto_entrada or 'consulta' in texto_entrada:
        return "🎯 Agende sua consultoria!\nLink: bit.ly/consultoria"
    
    elif 'obrigado' in texto_entrada or 'valeu' in texto_entrada or 'thanks' in texto_entrada:
        return "De nada! 😊 Estamos sempre aqui!"
    
    # Resposta padrão
    else:
        return """❓ Não entendi sua pergunta.
Digite 'menu' para ver opções ou 'atendente' para falar com uma pessoa."""

@app.route('/webhook', methods=['GET', 'POST'])
def webhook():
    """
    Webhook que recebe mensagens do WhatsApp
    """
    
    if request.method == 'GET':
        # Verificação do webhook
        token = request.args.get('hub.verify_token')
        challenge = request.args.get('hub.challenge')
        
        VERIFY_TOKEN = os.getenv('VERIFY_TOKEN', 'seu_token')
        
        if token == VERIFY_TOKEN:
            return challenge
        return 'Token inválido', 403
    
    if request.method == 'POST':
        # Processa mensagem recebida
        mensagem_entrada = request.values.get('Body', '').strip()
        numero_remetente = request.values.get('From', '').strip()
        
        print(f"📨 Mensagem recebida de {numero_remetente}: {mensagem_entrada}")
        
        # Processa a mensagem
        resposta = processar_mensagem(mensagem_entrada, numero_remetente)
        
        # Envia resposta
        response = MessagingResponse()
        response.message(resposta)
        
        print(f"✅ Resposta enviada: {resposta}\n")
        
        return str(response)

class GerenciadorContexto:
    def __init__(self):
        self.conversas = {}
        self.carregar_conversas()
    
    def carregar_conversas(self):
        try:
            if os.path.exists('conversas.json'):
                with open('conversas.json', 'r') as f:
                    self.conversas = json.load(f)
        except:
            self.conversas = {}
    
    def salvar_conversas(self):
        try:
            with open('conversas.json', 'w') as f:
                json.dump(self.conversas, f, ensure_ascii=False, indent=2)
        except:
            pass
    
    def obter_contexto(self, numero):
        if numero not in self.conversas:
            self.conversas[numero] = {
                'idioma': 'pt',
                'historico': [],
                'primeira_vez': True
            }
        return self.conversas[numero]
    
    def definir_idioma(self, numero, idioma):
        contexto = self.obter_contexto(numero)
        if idioma in IDIOMAS_SUPORTADOS:
            contexto['idioma'] = idioma
            self.salvar_conversas()
            return True
        return False

gerenciador = GerenciadorContexto()

def processar_mensagem_avancado(texto, numero_usuario=None):
    if numero_usuario:
        contexto = gerenciador.obter_contexto(numero_usuario)
        idioma = contexto['idioma']
    else:
        idioma = 'pt'
    
    strings = IDIOMAS_SUPORTADOS[idioma]
    texto = texto.lower().strip()
    
    if 'english' in texto or 'en' in texto:
        if numero_usuario:
            gerenciador.definir_idioma(numero_usuario, 'en')
        return "✅ Language changed to English"

        return "✅ Idioma cambiado a Español"
    
    elif any(word in texto for word in ['oi', 'olá', 'hola', 'hello']):

        return strings['preco']
    
    elif 'menu' in texto:

class AnalisadorSentimento:
    def __init__(self):
        self.positivas = ['ótimo', 'excelente', 'adorei', 'perfeito', 'bom']
        self.negativas = ['péssimo', 'horrível', 'odeio', 'ruim', 'problema']
    
    def analisar(self, texto):
        texto = texto.lower()
        score = 0
        
        for palavra in self.positivas:
            if palavra in texto:
                score += 1
        
        for palavra in self.negativas:
            if palavra in texto:
                score -= 1
        
        if score > 0:
            return 'positivo'
        elif score < 0:
            return 'negativo'
        else:
            return 'neutro'

analisador = AnalisadorSentimento()

# Use assim:
sentimento = analisador.analisar(mensagem_entrada)
print(f"Sentimento: {sentimento}")
