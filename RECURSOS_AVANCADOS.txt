from bot_avancado import processar_mensagem_avancado
from analisador_sentimentos import AnalisadorCompleto

analisador = AnalisadorCompleto()

@app.route('/webhook', methods=['POST'])
def webhook():
    msg = request.values.get('Body')
    numero = request.values.get('From')
    
    # Análise
    analise = analisador.analisar_completo(msg, numero)
    
    # Processar com contexto
    resposta = processar_mensagem_avancado(msg, numero)
    
    # Responder
    response = MessagingResponse()
    response.message(resposta)
    return str(response)

# No topo:
from bot_avancado import processar_mensagem_avancado
from analisador_sentimentos import AnalisadorCompleto

analisador = AnalisadorCompleto()

# Na função webhook():
@app.route('/webhook', methods=['POST'])
def webhook():
    mensagem = request.values.get('Body', '').strip()
    numero = request.values.get('From', '').strip()
    
    # Analisar
    analise = analisador.analisar_completo(mensagem, numero)
    
    # Log da análise
    print(f"📊 Análise: {analise['sentimento']} ({analise['categoria']})")
    
    # Processar com contexto
    resposta = processar_mensagem_avancado(mensagem, numero)
    
    # Responder
    response = MessagingResponse()
    response.message(resposta)
    
    return str(response)
